染分頁按鈕 function renderTabs() { const container = document.getElementById("tabsContainer"); container.innerHTML = ""; TABS.forEach(tab => { const btn = document.createElement("button"); btn.className = "tab-btn" + (tab.id === currentTab ? " active" : ""); btn.textContent = tab.label; btn.onclick = () => { currentTab = tab.id; applyThemeColor(); renderTabs(); renderPlaylists(); }; container.appendChild(btn); }); } // 渲染標籤篩選鈕（支援多選，再點一次取消，換分頁保留） function renderTags() { const container = document.getElementById("tagsContainer"); container.innerHTML = ""; ALL_TAGS.forEach(tag => { const btn = document.createElement("button"); btn.className = "tag-btn" + (selectedTags.has(tag) ? " active" : ""); btn.textContent = tag; btn.onclick = () => { if (selectedTags.has(tag)) { selectedTags.delete(tag); } else { selectedTags.add(tag); } renderTags(); renderPlaylists(); }; container.appendChild(btn); }); } // 渲染播放清單區塊與影片卡片 function renderPlaylists() { const container = document.getElementById("playlistContainer"); container.innerHTML = ""; // 1. 篩選目前分頁與已選標籤 const filtered = VIDEOS.filter(video => { if (video.tab !== currentTab) return false; if (selectedTags.size === 0) return true; return video.tags.some(t => selectedTags.has(t)); }); if (filtered.length === 0) { container.innerHTML = `
目前沒有符合此標籤條件的影片
`; return; } // 2. 依照播放清單名稱分組（維持資料出現順序） const groups = new Map(); filtered.forEach(video => { if (!groups.has(video.playlist)) { groups.set(video.playlist, []); } groups.get(video.playlist).push(video); }); // 3. 產生每個播放清單區塊 groups.forEach((videosInPlaylist, playlistName) => { // 依發布時間由新至舊排序 videosInPlaylist.sort((a, b) => parseDateValue(b.date) - parseDateValue(a.date)); const box = document.createElement("section"); box.className = "playlist-box"; const title = document.createElement("h2"); title.className = "playlist-title"; title.textContent = playlistName; box.appendChild(title); const row = document.createElement("div"); row.className = "video-row"; videosInPlaylist.forEach(video => { const card = document.createElement("a"); card.className = "video-card"; card.href = `https://www.youtube.com/watch?v=${video.youtubeId}`; card.target = "_blank"; card.rel = "noopener noreferrer"; const thumbUrl = `https://img.youtube.com/vi/${video.youtubeId}/hqdefault.jpg`; const tagsHtml = video.tags.map(t => `${t}`).join(""); card.innerHTML = `
${video.title}
${tagsHtml}
${video.date}
`; row.appendChild(card); }); box.appendChild(row); container.appendChild(box); }); } // 初始化 applyThemeColor(); renderTabs(); renderTags(); renderPlaylists();
